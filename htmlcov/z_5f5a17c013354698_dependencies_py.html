55def _resolve_test_auth_user(token: str, db: Session) -> User: 

56 """Достать сид-юзера по токену dev/test-байпаса. 

57 

58 Принимаем ТОЛЬКО сид-юзеров (`is_test`): даже с валидным секретом токен на 

59 реальный аккаунт не резолвится. Битый UUID/несуществующий/не-тест → 401 без 

60 утечки, работают ли какие-то юзеры. 

61 """ 

62 raw_id = token.split(':', 1)[1] 

63 try: 

64 user_id = UUID(raw_id) 

65 except ValueError: 

66 raise HTTPException( 

67 status_code=HTTP_401_UNAUTHORIZED, detail='Not authenticated' 

68 ) from None 

69 user = db.execute( 

70 select(User).where(User.id == user_id, User.is_test.is_(True)) 

71 ).scalar_one_or_none() 

72 if not user: 

73 raise HTTPException( 

74 status_code=HTTP_401_UNAUTHORIZED, detail='Not authenticated' 

75 ) 

76 return user

88def _authenticate(request: Request, db: Session) -> User: 

89 token = request.headers.get('Authorization') 

90 if not token: 

91 raise HTTPException( 

92 status_code=HTTP_401_UNAUTHORIZED, detail='Not authenticated' 

93 ) 

94 

95 # dev/test-байпас (фича 0009): работает и в проде-подобной среде, но только 

96 # для сид-юзеров. Гейтится наличием секрета, не `IS_DEBUG`. 

97 if _is_test_auth_token(token): 

98 return _resolve_test_auth_user(token, db) 

99 

100 try: 

101 decoded_token = verify_id_token(token) 

102 except ExpiredIdTokenError: 

103 raise HTTPException(HTTP_401_UNAUTHORIZED, 'Token expired') from None 

104 except InvalidIdTokenError: 

105 raise HTTPException(HTTP_401_UNAUTHORIZED, 'Invalid token') from None 

106 uid = decoded_token['uid'] 

107 user = db.execute(select(User).where(User.firebase_uid == uid)).scalar_one_or_none() 

108 if not user: 

109 raise HTTPException( 

110 status_code=HTTP_401_UNAUTHORIZED, detail='Not authenticated' 

111 ) 

112 return user

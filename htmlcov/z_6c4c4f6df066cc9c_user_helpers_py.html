24def guess_image_extension(content: bytes) -> str: 

25 """Расширение картинки по сигнатуре (magic bytes), с ведущей точкой. 

26 

27 Нужно, чтобы файлы на диске имели расширение и StaticFiles/nginx отдавали их 

28 с корректным Content-Type (иначе — application/octet-stream). Пустая строка, 

29 если тип не распознан (сохраним без расширения, как раньше). 

30 """ 

31 if content.startswith(b'\x89PNG\r\n\x1a\n'): 

32 return '.png' 

33 if content.startswith(b'\xff\xd8\xff'): 

34 return '.jpg' 

35 if content.startswith((b'GIF87a', b'GIF89a')): 

36 return '.gif' 

37 if content[:4] == b'RIFF' and content[8:12] == b'WEBP': 

38 return '.webp' 

39 return ''

42def save_profile_image_bytes(user: User, content: bytes, *, is_custom: bool) -> None: 

43 """Сохранить байты аватарки на диск и проставить пользователю ссылку. 

44 

45 Единая точка записи фото на диск — используется и ручной загрузкой 

46 (`set_profile_image`, `is_custom=True`), и бэкфиллом соц-аватарок на диск 

47 (`is_custom=False`). URL строим из доверенного `FRONTEND_URL`, а не из Host 

48 запроса (host-header injection / stored URL poisoning). Файл сохраняем с 

49 расширением по типу картинки — чтобы отдавался с корректным Content-Type. 

50 """ 

51 PROFILE_IMAGES_DIR.mkdir(exist_ok=True, parents=True) 

52 extension = guess_image_extension(content) 

53 file_name = f'profile_image_user_{user.id}_{datetime.now().isoformat()}{extension}' 

54 file_path = PROFILE_IMAGES_DIR / file_name 

55 file_path.write_bytes(content) 

56 related_media_path = file_path.relative_to(settings.MEDIA_ROOT) 

57 user.photo_url = f'{settings.FRONTEND_URL}/media/{related_media_path}' 

58 user.photo_path = str(file_path) 

59 user.photo_is_custom = is_custom

62def upscale_google_avatar_url(url: str) -> str: 

63 """Поднять разрешение Google-аватарки до TARGET_AVATAR_SIZE px. 

64 

65 Firebase отдаёт Google-фото с зашитым в URL размером `=s96-c` (96px), из-за 

66 чего аватарки пикселят при показе крупнее. Google по тому же URL отдаёт вплоть 

67 до оригинала, если заменить токен размера. Не-Google URL (VK и пр.) — как есть. 

68 """ 

69 if 'googleusercontent.com' not in url: 

70 return url 

71 # Токен размера у Google-аватарок: `=sNN` или `=sNN-c` (обычно в конце URL). 

72 upscaled, replaced = re.subn(r'=s\d+(-c)?', f'=s{TARGET_AVATAR_SIZE}-c', url) 

73 if replaced: 

74 return upscaled 

75 return f'{url}=s{TARGET_AVATAR_SIZE}-c'

78def download_avatar_bytes(url: str, client: httpx.Client | None = None) -> bytes | None: 

79 """Скачать аватарку (Google — в высоком разрешении). None при ошибке. 

80 

81 Ошибку скачивания логируем и глушим (возвращаем None): вызывающий решает, 

82 что делать (обнулить битую ссылку / оставить текущее фото). 

83 """ 

84 own_client = client is None 

85 if client is None: 

86 client = httpx.Client(timeout=AVATAR_DOWNLOAD_TIMEOUT_SECONDS) 

87 try: 

88 response = client.get(upscale_google_avatar_url(url), follow_redirects=True) 

89 response.raise_for_status() 

90 return response.content 

91 except httpx.HTTPError as exc: 

92 logger.warning('Не удалось скачать аватарку url={url}: {exc}', url=url, exc=exc) 

93 return None 

94 finally: 

95 if own_client: 

96 client.close()

122def get_annotated_users( 

123 db: Session, 

124 current_user: User, 

125 outer_users: Select[tuple[User]] | Sequence[User] | None = None, 

126) -> list[AnnotatedOtherUserSchema]: 

127 query = select( 

128 User, 

129 User.followed_by.any(User.id == current_user.id).label('followed_by_me'), 

130 User.follows.any(User.id == current_user.id).label('follows_me'), 

131 ) 

132 if isinstance(outer_users, Select): 

133 user_ids = [user.id for user in db.execute(outer_users).scalars()] 

134 query = query.where(User.id.in_(user_ids)) 

135 elif outer_users is not None: 

136 user_ids = [user.id for user in outer_users] 

137 query = query.where(User.id.in_(user_ids)) 

138 values = db.execute(query).all() 

139 for user, followed_by_me, follows_me in values: 

140 user.followed_by_me = followed_by_me 

141 user.follows_me = follows_me 

142 return [AnnotatedOtherUserSchema.model_validate(val[0]) for val in values]

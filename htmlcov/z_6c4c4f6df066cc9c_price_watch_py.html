1"""Наблюдатель цен и наличия (фича 0010) — чистая часть обхода. 

2 

3Обход разрезан так, чтобы всё, кроме HTTP-похода, тестировалось на сохранённом 

4ответе магазина без моков: 

5 

6 выбор целей (`select_watch_targets`) → батчи (`batched`) → 

7 запрос (`fetch_wb_cards`, клиент инъецируется) → ответ в Pydantic → 

8 наблюдения (`observe_batch`, батч + ответ) → запись истории 

9 (`save_observations`) → обновление магазинных хотелок (`apply_observations`). 

10 

11Конвертации нужен именно батч, а не только ответ: состояние «артикул исчез» — это 

12ОТСУТСТВИЕ товара в ответе, а отсутствие видно лишь зная, что запрашивали. 

13 

14Та же конвертация обслуживает свежий запрос по одной ссылке из пользовательского 

15сценария (фича 0011: превью, сохранение, кнопка) — `fetch_fresh_observation`. 

16"""

74def select_watch_targets(db: Session) -> list[WatchTarget]: 

75 """Активные хотелки со ссылкой на WB, из которой читается артикул. 

76 

77 Хотелка без ссылки или на другой магазин просто не попадает в обход — 

78 это норма, не ошибка. Архивные выбывают (🟡 Q2 intent'а). 

79 """ 

80 wishes = db.execute( 

81 select(Wish.id, Wish.link).where(Wish.link.isnot(None), ~Wish.is_archived) 

82 ).all() 

83 targets = [] 

84 for wish_id, link in wishes: 

85 parsed = parse_wildberries_link(link) 

86 if parsed is None: 

87 continue 

88 targets.append(WatchTarget(wish_id, *parsed)) 

89 return targets

92def batched(targets: Sequence[WatchTarget], size: int) -> Iterator[list[WatchTarget]]: 

93 for start in range(0, len(targets), size): 

94 yield list(targets[start : start + size])

97def fetch_wb_cards(skus: Sequence[int], client: httpx.Client) -> WbCardResponseSchema: 

98 """Один запрос к WB за батчем артикулов. Единственная сетевая функция обхода.""" 

99 response = client.get( 

100 WB_CARD_API_URL, 

101 params={**WB_CARD_API_PARAMS, 'nm': ';'.join(str(sku) for sku in skus)}, 

102 ) 

103 response.raise_for_status() 

104 return WbCardResponseSchema.model_validate(response.json())

111@dataclass(frozen=True) 

112class ProductObservation: 

113 """Результат одного наблюдения: статус, цена (только при `ok`) и признак 

114 «цена — минимум среди размеров» (ссылка без размера у многоразмерного 

115 товара; UI рисует «от»).""" 

116 

117 status: PriceObservationStatus 

118 price: WbPriceSchema | None = None 

119 is_minimum: bool = False 

120 

121 @property 

122 def product_price(self) -> Decimal | None: 

123 return _kopecks_to_rubles(self.price.product) if self.price else None 

124 

125 @property 

126 def basic_price(self) -> Decimal | None: 

127 return _kopecks_to_rubles(self.price.basic) if self.price else None

130def observe_product( 

131 size_option_id: int | None, product: WbProductSchema | None 

132) -> ProductObservation: 

133 """Статус и цена товара по карточке WB. 

134 

135 Размер из ссылки известен → смотрим ровно его; размера в карточке больше 

136 нет — считаем товар исчезнувшим (то, что юзер откладывал, купить нельзя). 

137 Размера в ссылке нет → минимальная `product`-цена среди размеров в наличии; 

138 в наличии ни одного — распродан. 

139 """ 

140 if product is None: 

141 return ProductObservation(PriceObservationStatus.gone) 

142 if size_option_id is not None: 

143 size = next((s for s in product.sizes if s.option_id == size_option_id), None) 

144 if size is None: 

145 return ProductObservation(PriceObservationStatus.gone) 

146 if size.price is None: 

147 return ProductObservation(PriceObservationStatus.sold_out) 

148 return ProductObservation(PriceObservationStatus.ok, size.price) 

149 in_stock = [s.price for s in product.sizes if s.price is not None] 

150 if not in_stock: 

151 return ProductObservation(PriceObservationStatus.sold_out) 

152 return ProductObservation( 

153 PriceObservationStatus.ok, 

154 min(in_stock, key=lambda p: p.product), 

155 is_minimum=len(product.sizes) > 1, 

156 )

159def observe_batch( 

160 batch: Sequence[WatchTarget], response: WbCardResponseSchema 

161) -> list[tuple[WatchTarget, ProductObservation]]: 

162 """Батч + ответ WB → наблюдение по каждой цели (отсутствие в ответе = gone).""" 

163 products = {product.id: product for product in response.products} 

164 return [ 

165 (target, observe_product(target.size_option_id, products.get(target.sku))) 

166 for target in batch 

167 ]

198def sync_wish_with_store( 

199 wish: Wish, observation: ProductObservation, observed_at: datetime 

200) -> None: 

201 """Применить наблюдение к магазинной хотелке. 

202 

203 Наличие и давность — всегда; цена и «от» — только когда товар в наличии: 

204 магазин цену никогда не обнуляет, при распродано/исчез остаётся последняя 

205 известная (решение продукта, intent 0011). 

206 """ 

207 wish.store_availability = observation.status 

208 wish.store_observed_at = observed_at 

209 if observation.status == PriceObservationStatus.ok and observation.product_price: 

210 # Контракт отдаёт цену целым числом рублей, копейки отбрасываются; в 

211 # истории наблюдений цена остаётся точной. 

212 wish.price = observation.product_price.to_integral_value(rounding=ROUND_DOWN) 

213 wish.price_is_minimum = observation.is_minimum

216def apply_observations( 

217 db: Session, 

218 observed: Sequence[tuple[WatchTarget, ProductObservation]], 

219 observed_at: datetime, 

220) -> int: 

221 """Обновить хотелки батча с магазинной ценой; возвращает число обновлённых. 

222 

223 Ручные (`manual`) не трогаем — юзер поменял цену, магазин для него молчит; 

224 история наблюдений по ним всё равно пишется (прибор 0010). 

225 """ 

226 by_wish_id = {target.wish_id: observation for target, observation in observed} 

227 wishes = db.scalars( 

228 select(Wish).where( 

229 Wish.id.in_(by_wish_id), Wish.price_source == PriceSource.shop 

230 ) 

231 ).all() 

232 for wish in wishes: 

233 sync_wish_with_store(wish, by_wish_id[wish.id], observed_at) 

234 db.commit() 

235 return len(wishes)

238class EmptyStoreResponseError(Exception): 

239 """Магазин ответил 200, но без запрошенной карточки. 

240 

241 Для одиночного свежего запроса это неотличимо от «артикул исчез», а WB 

242 временами отдаёт пустой `products` на живой товар (прод, 2026-09-12: 

243 превью без цены при in-stock артикуле, через минуты — цена есть). Считать 

244 такое «исчез» нельзя: хотелка получила бы «товара больше нет» до ночного 

245 обхода. Поэтому — сбой магазина (тихо/`502`), а «исчез» ставит только обход, 

246 где отсутствие в батче — надёжный сигнал. 

247 """

250def fetch_fresh_observation( 

251 link: str, client: httpx.Client 

252) -> ProductObservation | None: 

253 """Свежий запрос к магазину по одной ссылке (фича 0011). 

254 

255 None — ссылка не на поддерживаемый магазин (спрашивать нечего). Сетевые 

256 ошибки, чужой формат и пустой ответ идут наверх (`httpx.HTTPError`, 

257 `pydantic.ValidationError`, `EmptyStoreResponseError`) — вызывающий решает, 

258 тихо это или `502`. 

259 """ 

260 parsed = parse_wildberries_link(link) 

261 if parsed is None: 

262 return None 

263 sku, size_option_id = parsed 

264 response = fetch_wb_cards([sku], client) 

265 products = {product.id: product for product in response.products} 

266 if sku not in products: 

267 raise EmptyStoreResponseError(f'карточки {sku} нет в ответе магазина') 

268 return observe_product(size_option_id, products[sku])

271def record_fresh_observation( 

272 db: Session, wish: Wish, observation: ProductObservation, observed_at: datetime 

273) -> None: 

274 """Свежее наблюдение из пользовательского сценария: обновить хотелку и 

275 дописать историю (если за эти сутки строки ещё нет — первое наблюдение за 

276 сутки остаётся истиной прибора). Коммит — на вызывающем.""" 

277 sync_wish_with_store(wish, observation, observed_at) 

278 parsed = parse_wildberries_link(wish.link or '') 

279 assert parsed is not None, 'наблюдение бывает только у WB-ссылки' 

280 target = WatchTarget(wish.id, *parsed) 

281 db.execute( 

282 pg_insert(WishPriceObservation) 

283 .values([observation_row(target, observation, observed_at.date())]) 

284 .on_conflict_do_nothing(index_elements=['wish_id', 'observed_date']) 

285 )

288def save_observations(db: Session, observations: Sequence[dict]) -> int: 

289 """Записать батч наблюдений; возвращает число реально вставленных строк. 

290 

291 `ON CONFLICT DO NOTHING` по `(wish_id, observed_date)`: первое наблюдение за 

292 сутки — истина, повторный запуск в те же сутки не перетирает его, а лишь 

293 дозаполняет хотелки, чей батч днём упал (бесплатный ретрай). 

294 """ 

295 if not observations: 

296 return 0 

297 result = db.execute( 

298 pg_insert(WishPriceObservation) 

299 .values(list(observations)) 

300 .on_conflict_do_nothing(index_elements=['wish_id', 'observed_date']) 

301 .returning(WishPriceObservation.wish_id) 

302 ) 

303 inserted = len(result.all()) 

304 db.commit() 

305 return inserted

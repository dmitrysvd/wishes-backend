1"""httpx-транспорт поверх curl_cffi с TLS-отпечатком Chrome. 

2 

3WAF WB (`server: wbaas`) фильтрует по отпечатку TLS-рукопожатия (JA3): 

4отпечаток питоновского `ssl` + httpx у него в чёрном списке, заголовки не 

5влияют — 403 приходит до первого байта HTTP. curl_cffi воспроизводит 

6ClientHello Chrome, и запрос выглядит как браузерный. Интерфейс httpx 

7сохранён, чтобы код обхода и его тесты на `httpx.MockTransport` не зависели от 

8библиотеки под капотом. 

9""" 

10 

11from typing import cast 

12 

13import httpx 

14from curl_cffi import requests as curl_requests 

15from curl_cffi.requests.session import HttpMethod 

16 

17# Профиль curl_cffi без версии — библиотека подставляет актуальный Chrome. 

18IMPERSONATE_BROWSER = 'chrome' 

19# Заголовки, которые httpx.Client ставит сам. Их не пробрасываем: curl_cffi 

20# подставляет браузерные, согласованные с отпечатком (иначе UA python-httpx). 

21HTTPX_DEFAULT_HEADERS = frozenset( 

22 {'host', 'accept', 'accept-encoding', 'connection', 'user-agent'} 

23) 

24 

25 

26class BrowserTransport(httpx.BaseTransport): 

27 def __init__(self, timeout: float) -> None: 

28 self._session = curl_requests.Session( 

29 impersonate=IMPERSONATE_BROWSER, timeout=timeout 

30 ) 

31 

32 def handle_request(self, request: httpx.Request) -> httpx.Response: 

33 headers = { 

34 name: value 

35 for name, value in request.headers.items() 

36 if name.lower() not in HTTPX_DEFAULT_HEADERS 

37 } 

38 response = self._session.request( 

39 cast(HttpMethod, request.method), 

40 str(request.url), 

41 headers=headers, 

42 data=request.content or None, 

43 ) 

44 return httpx.Response( 

45 status_code=response.status_code, 

46 headers=list(response.headers.multi_items()), 

47 content=response.content, 

48 request=request, 

49 ) 

50 

51 def close(self) -> None: 

52 self._session.close()

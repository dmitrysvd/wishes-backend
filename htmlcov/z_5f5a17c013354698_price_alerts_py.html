4превращение суточных наблюдений обхода (0010) в один пуш в сутки автору хотелки. 

5Продуктовые правила — в intent 0013 (шина); их сторона «что видит клиент» 

6(payload, deep link) — `x-push-payload` у `POST /push/opened`. Сами правила —

103def seen_price(db: Session, wish: Wish, last_activity: date | None) -> Decimal | None: 

104 """База «видел» без участия клиента. 

105 

106 Что позже: сохранённая база (пуш / кнопка / включение группы) либо цена 

107 наблюдения на последний день активности юзера (карточка показывала его). 

108 Ни того ни другого — цена, с которой хотелка началась (первое наблюдение: 

109 оно пишется при сохранении с WB-ссылкой). Наблюдение без цены 

110 (распродано/нет) базой быть не может. 

111 """ 

112 base_price, base_at = wish.alert_base_price, wish.alert_base_at 

113 activity_obs = ( 

114 _observation_on_or_before(db, wish.id, last_activity) 

115 if last_activity is not None 

116 else None 

117 ) 

118 if activity_obs is not None and activity_obs.product_price is None: 

119 activity_obs = None 

120 if base_price is not None and base_at is not None: 

121 if activity_obs is None or activity_obs.observed_date <= base_at.date(): 

122 return base_price 

123 if activity_obs is not None: 

124 return activity_obs.product_price 

125 first = _first_priced_observation(db, wish.id) 

126 return first.product_price if first is not None else None

129def _latest_two( 

130 db: Session, wish_id: UUID 

131) -> tuple[WishPriceObservation | None, WishPriceObservation | None]: 

132 rows = db.scalars( 

133 select(WishPriceObservation) 

134 .where(WishPriceObservation.wish_id == wish_id) 

135 .order_by(WishPriceObservation.observed_date.desc()) 

136 .limit(2) 

137 ).all() 

138 latest = rows[0] if rows else None 

139 previous = rows[1] if len(rows) > 1 else None 

140 return latest, previous

143def detect_alert( 

144 db: Session, wish: Wish, today: date, last_activity: date | None 

145) -> WishAlert | None: 

146 """Сработала ли хотелка по сегодняшнему наблюдению обхода. 

147 

148 Только по наблюдению за `today`: обход упал — событий нет, задним числом не 

149 догоняем. «Снова в наличии» — предыдущее наблюдение распродано/нет; иначе 

150 «подешевело» — цена ниже базы «видел» на ≥ порога. Рост цены — молчим. 

151 """ 

152 latest, previous = _latest_two(db, wish.id) 

153 if ( 

154 latest is None 

155 or latest.observed_date != today 

156 or latest.status != PriceObservationStatus.ok 

157 or latest.product_price is None 

158 ): 

159 return None 

160 price = latest.product_price 

161 if previous is not None and previous.status in UNAVAILABLE: 

162 # Возврат в наличие: событие само по себе, без сравнения с базой. Пока 

163 # триггер выключен — молчим (сравнивать с базой до распродажи не по чему). 

164 if not PRICE_ALERT_AVAILABILITY_ENABLED: 

165 return None 

166 return WishAlert(wish, PriceAlertTrigger.availability, price, None) 

167 # База есть всегда: сегодняшнее наблюдение с ценой — уже кандидат в «первое». 

168 base = seen_price(db, wish, last_activity) 

169 assert base is not None 

170 if price <= base * (1 - PRICE_ALERT_DROP_RATIO): 

171 return WishAlert(wish, PriceAlertTrigger.price, price, base) 

172 return None

175def rubles(value: Decimal) -> str: 

176 """`2700` → `2 700 ₽`: целые рубли, пробел-разделитель тысяч (контракт).""" 

177 whole = int(value.to_integral_value(rounding=ROUND_DOWN)) 

178 return f'{whole:,}'.replace(',', ' ') + ' ₽'

181def _things(count: int) -> str: 

182 """«2 вещи», «5 вещей», «21 вещь» — склонение по последним цифрам.""" 

183 tail, tail2 = count % 10, count % 100 

184 if tail == 1 and tail2 != 11: 

185 word = 'вещь' 

186 elif 2 <= tail <= 4 and not 12 <= tail2 <= 14: 

187 word = 'вещи' 

188 else: 

189 word = 'вещей' 

190 return f'{count} {word}'

200def _one_line(alert: WishAlert) -> tuple[str, str]: 

201 """(заголовок, тело) одной хотелки — тексты из контракта.""" 

202 name = f'„{short_name(alert.wish.name)}“' 

203 if alert.trigger == PriceAlertTrigger.availability: 

204 return f'{name} снова в наличии', f'{rubles(alert.price)} на WB' 

205 assert alert.was_price is not None 

206 return ( 

207 f'{name} подешевела', 

208 f'{rubles(alert.price)} вместо {rubles(alert.was_price)}', 

209 )

212def build_message( 

213 user: User, alerts: list[WishAlert] 

214) -> tuple[str, str, str, PriceAlertTrigger]: 

215 """(title, body, link, trigger) дайджеста по контракту `POST /push/opened`.""" 

216 first = alerts[0] 

217 if len(alerts) == 1: 

218 title, body = _one_line(first) 

219 link = f'{settings.FRONTEND_URL}/wish?wishId={first.wish.id}' 

220 return title, body, link, first.trigger 

221 triggers = {alert.trigger for alert in alerts} 

222 trigger = triggers.pop() if len(triggers) == 1 else PriceAlertTrigger.mixed 

223 _, first_body = _one_line(first) 

224 # Строка первой хотелки: «„Название“: 2 700 ₽ вместо 3 000 ₽». 

225 body = f'„{short_name(first.wish.name)}“: {first_body}' 

226 title = f'{_things(len(alerts))} из списка подешевели или вернулись в наличие' 

227 return title, body, get_user_deep_link(user), trigger

230def already_sent_today(db: Session, user_id: UUID, today: date) -> bool: 

231 """Один пуш по складу в календарные сутки UTC — по логу отправок. 

232 

233 `sent_at` в логе — naive `datetime.now()` (сервер живёт в UTC), поэтому и 

234 граница суток naive.""" 

235 day_start = datetime.combine(today, datetime.min.time()) 

236 return ( 

237 db.scalars( 

238 select(PushSendingLog.id).where( 

239 PushSendingLog.reason == PushReason.PRICE_ALERT, 

240 PushSendingLog.target_user_id == user_id, 

241 PushSendingLog.sent_at >= day_start, 

242 ) 

243 ).first() 

244 is not None 

245 )

248@dataclass(frozen=True) 

249class UserDigest: 

250 """Что ушло бы юзеру сегодня: сработавшие хотелки и собранный текст.""" 

251 

252 user: User 

253 alerts: list[WishAlert] 

254 title: str 

255 body: str 

256 link: str 

257 trigger: PriceAlertTrigger

259 

260def collect_digests(today: date) -> list[UserDigest]: 

261 """Чтение без побочных эффектов: кому и что ушло бы за `today`. 

262 

263 Общая часть крона и dry-run. Дедуп «один в сутки» — здесь (по логу); 

264 выключенная группа — нет: её отсекает `send_push`, а dry-run помечает. 

265 """ 

266 with SessionLocal() as db: 

267 users = db.scalars(select(User).where(User.can_receive_push)) 

268 candidates = [(user, store_wishes(user)) for user in users] 

269 digests: list[UserDigest] = [] 

270 for user, wishes in candidates: 

271 if not wishes: 

272 continue 

273 with SessionLocal() as db: 

274 if already_sent_today(db, user.id, today): 

275 continue 

276 last_activity = _last_activity_date(db, user.id) 

277 alerts = [ 

278 alert 

279 for wish in wishes 

280 if (alert := detect_alert(db, wish, today, last_activity)) 

281 ] 

282 if not alerts: 

283 continue 

284 title, body, link, trigger = build_message(user, alerts) 

285 digests.append(UserDigest(user, alerts, title, body, link, trigger)) 

286 return digests

289def send_price_alerts(today: date | None = None) -> int: 

290 """Крон в полдень после суточного обхода: дайджест по складу каждому автору. 

291 

292 База «видел» сдвигается на цену из пуша только если FCM принял сообщение; 

293 выключенная группа `prices` отсекается в `send_push` — базу не двигает и 

294 события не копит. Возвращает число отправленных пушей. 

295 """ 

296 if not PRICE_ALERT_ENABLED: 

297 logger.info('Пуши по складу выключены (PRICE_ALERT_ENABLED) — пропуск') 

298 return 0 

299 today = today or utc_now().date() 

300 sent_total = 0 

301 for digest in collect_digests(today): 

302 user = digest.user 

303 outcome = send_push( 

304 [user], 

305 digest.title, 

306 digest.body, 

307 reason=PushReason.PRICE_ALERT, 

308 link=digest.link, 

309 trigger=digest.trigger, 

310 with_delivery_id=True, 

311 ) 

312 sent_total += len(outcome.sent_user_ids) 

313 if user.id not in outcome.accepted_user_ids: 

314 continue 

315 now = utc_now() 

316 with SessionLocal() as db: 

317 for alert in digest.alerts: 

318 wish = db.get(Wish, alert.wish.id) 

319 if wish is not None: 

320 wish.alert_base_price = alert.price 

321 wish.alert_base_at = now 

322 db.commit() 

323 logger.info(f'Пуши по складу: отправлено {sent_total}') 

324 return sent_total

327def dry_run_report(today: date | None = None) -> str: 

328 """Сухой прогон: что ушло бы, без отправки, лога и сдвига базы. 

329 

330 Персональных данных сверх названий хотелок нет: юзер — только id. Юзеры с 

331 выключенной группой показаны с пометкой — реальный крон их пропустит. 

332 """ 

333 today = today or utc_now().date() 

334 digests = collect_digests(today) 

335 with SessionLocal() as db: 

336 opted_out = disabled_user_ids( 

337 db, (d.user.id for d in digests), PushReason.PRICE_ALERT 

338 ) 

339 lines = [ 

340 f'dry-run за {today}: юзеров {len(digests)} ' 

341 f'(из них с выключенной группой {len(opted_out)}), ' 

342 f'строк {sum(len(d.alerts) for d in digests)}', 

343 ] 

344 by_trigger: dict[str, int] = {} 

345 for digest in digests: 

346 by_trigger[digest.trigger.value] = by_trigger.get(digest.trigger.value, 0) + 1 

347 lines.append( 

348 'по типу пуша: ' + ', '.join(f'{k}={v}' for k, v in sorted(by_trigger.items())) 

349 ) 

350 for digest in digests: 

351 flag = ' [группа выключена]' if digest.user.id in opted_out else '' 

352 lines.append(f'--- user {digest.user.id}{flag} → {digest.trigger.value}') 

353 for alert in digest.alerts: 

354 was = rubles(alert.was_price) if alert.was_price is not None else '—' 

355 lines.append( 

356 f' {alert.trigger.value:<12} {alert.wish.name!r}: ' 

357 f'{was} → {rubles(alert.price)}' 

358 ) 

359 lines.append(f' push: {digest.title} / {digest.body}') 

360 return '\n'.join(lines)

363def mark_seen(wish: Wish, price: Decimal | None, at: datetime) -> None: 

364 """Юзер «видел» цену: кнопка «актуальная с WB» / включение группы.

366 if price is None:

372def reset_seen_for_user(db: Session, user: User) -> None: 

373 """Включение группы `prices`: база = текущая цена карточки у всех активных

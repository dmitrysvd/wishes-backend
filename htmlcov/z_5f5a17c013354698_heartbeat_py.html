1"""Heartbeat-отметки фоновых процессов. 

2 

3Фоновые задачи (ночной бэкап, планировщик) не слушают порт — снаружи их не 

4опросишь, и молчаливая смерть такой задачи ничем себя не проявляет. Поэтому 

5задача сама отмечается в файле при каждом успешном прогоне, а приложение отдаёт 

6свежесть отметки обычной HTTP-ручкой (`/health/heartbeat/{name}`). Так внешний 

7uptime-монитор ловит зависший процесс тем же механизмом, что и падение сайта. 

8""" 

9 

10import time 

11from pathlib import Path 

12 

13 

14def touch(name: str, directory: Path) -> None: 

15 """Отметить успешный прогон процесса `name`.""" 

16 directory.mkdir(exist_ok=True, parents=True) 

17 (directory / name).touch()

19 

20def age_seconds(name: str, directory: Path) -> float | None: 

21 """Возраст отметки в секундах. None — отметки нет вовсе.""" 

22 path = directory / name 

23 if not path.exists(): 

24 return None 

25 return time.time() - path.stat().st_mtime

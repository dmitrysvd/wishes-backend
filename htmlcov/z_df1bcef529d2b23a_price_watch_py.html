1"""Суточный обход цен и наличия по хотелкам (фича 0010). 

2 

3Best-effort: упавший батч (сеть, 5xx, чужой формат ответа) — пропущенное 

4наблюдение и `warning`, обход идёт дальше; коммит по батчам, чтобы удачные не 

5пропали из-за неудачных. `error` — только если весь обход не дал ни одного 

6наблюдения: это уже смена формата или бан, а не флуктуация. 

7Прибор не имеет права влиять ни на что живое. 

8""" 

9 

10import time 

11from datetime import date 

12 

13import httpx 

14 

15from app.constants import PRICE_WATCH_BATCH_PAUSE_SECONDS, PRICE_WATCH_BATCH_SIZE 

16from app.db import SessionLocal 

17from app.helpers.browser_transport import BrowserTransport 

18from app.helpers.price_watch import ( 

19 apply_observations, 

20 batched, 

21 fetch_wb_cards, 

22 observation_row, 

23 observe_batch, 

24 save_observations, 

25 select_watch_targets, 

26) 

27from app.logging import logger 

28from app.utils import utc_now 

29 

30# Таймаут одного запроса к магазину. 

31WB_REQUEST_TIMEOUT = 20 

32 

33 

34def crawl( 

35 client: httpx.Client, 

36 observed_date: date | None = None, 

37 batch_size: int = PRICE_WATCH_BATCH_SIZE, 

38 pause_seconds: float = PRICE_WATCH_BATCH_PAUSE_SECONDS, 

39) -> int: 

40 """Обойти все цели одним проходом; возвращает число записанных наблюдений. 

41 

42 Клиент, дата и пауза параметризованы ради тестируемости без моков 

43 (`httpx.MockTransport`, фиксированная дата, нулевая пауза). 

44 """ 

45 observed_at = utc_now() 

46 observed_date = observed_date or observed_at.date() 

47 with SessionLocal() as db: 

48 targets = select_watch_targets(db) 

49 logger.info(f'Обход цен: {len(targets)} хотелок, дата {observed_date}') 

50 saved = 0 

51 failed_batches = 0 

52 for index, batch in enumerate(batched(targets, batch_size)): 

53 if index: 

54 time.sleep(pause_seconds) 

55 try: 

56 response = fetch_wb_cards([t.sku for t in batch], client) 

57 observed = observe_batch(batch, response) 

58 observations = [ 

59 observation_row(target, observation, observed_date) 

60 for target, observation in observed 

61 ] 

62 with SessionLocal() as db: 

63 saved += save_observations(db, observations) 

64 # Фича 0011: магазинные хотелки получают цену/наличие из обхода. 

65 apply_observations(db, observed, observed_at) 

66 except Exception as error: 

67 # Пропущенное наблюдение, не ошибка: дырка в истории допустима. 

68 failed_batches += 1 

69 logger.warning(f'Обход цен: батч #{index} пропущен: {error!r}') 

70 if targets and not saved and failed_batches: 

71 logger.error( 

72 f'Обход цен не дал ни одного наблюдения: {failed_batches} батчей упало' 

73 ) 

74 logger.info(f'Обход цен: записано {saved}, батчей упало {failed_batches}') 

75 return saved 

76 

77 

78def main() -> None: 

79 # С отпечатком обычного httpx WB отвечает 403 на всё — см. BrowserTransport. 

80 with httpx.Client(transport=BrowserTransport(timeout=WB_REQUEST_TIMEOUT)) as client: 

81 crawl(client)

83 

84if __name__ == '__main__': 

85 main()
